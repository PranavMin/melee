# Tournament module hook table (tools/build_module.py).
# One patch per line, applied to the VANILLA Melee 1.02 DOL in RAM by the loaders
# (Nintendont kernel / Ishiiruka fork) after the module blob is copied in:
#   ptr    <addr> <module symbol>   write the symbol's address at addr
#   branch <addr> <module symbol>   write "b <symbol>" at addr (function hijack)
#   word   <addr> <hex>             write a literal 32-bit word
# Addresses are vanilla (config/GALE01/symbols.txt); the builder verifies each
# against orig/GALE01/sys/main.dol where noted.

# mnmain.c menu-kind table mn_803EB6B0 (rows of 0x14: anim_loop, start_frame,
# description_indices, selection_count, think).
ptr    0x803EB6C0 mnTourney_MainMenuThink   # row 0 (MAIN) think (vanilla mn_8022DB10)
# The Tournament menu hijacks row 3 (MENU_KIND_TOY, "Trophies"): the vanilla
# table cannot grow, and a kiosk has no use for the trophy menu - picking
# "Trophies" on the main menu now opens the set list. anim_loop/start_frame
# are zeroed so the panel keeps the plain frame instead of the green
# Trophies one.
word   0x803EB6EC 0x00000000                # row 3 anim_loop = NULL (was the Trophies panel anim)
word   0x803EB6F0 0x00000000                # row 3 start_frame = 0.0f (frame 60 = green Trophies frame)
# The menu PANEL's title/frame animation per kind lives in a second table,

# idle). Kind 3's rows are the green "Trophies" frames (600-799); the old

# 3 gets exactly those words: {0,49,-0.1} {50,250,50} {0,10,-0.1}.

word   0x803EAEFC 0x42440000                # [3][0].end    49

word   0x803EAF04 0x42480000                # [3][1].start  50
word   0x803EAF08 0x437A0000                # [3][1].end    250
word   0x803EAF0C 0x42480000                # [3][1].loop   50
word   0x803EAF10 0x00000000                # [3][2].start  0
word   0x803EAF14 0x41200000                # [3][2].end    10
word   0x803EAF18 0xBDCCCCCD                # [3][2].loop   -0.1
